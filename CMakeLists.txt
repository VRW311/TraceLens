cmake_minimum_required(VERSION 3.16)
project(TraceLens VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(tracelens_core core/engine.cpp)
target_include_directories(tracelens_core PUBLIC core)
target_compile_options(tracelens_core PRIVATE -Wall -Wextra -Wpedantic)
if(EMSCRIPTEN)
  add_executable(tracelens_wasm core/wasm.cpp)
  target_link_libraries(tracelens_wasm PRIVATE tracelens_core)
  set_target_properties(tracelens_wasm PROPERTIES OUTPUT_NAME tracelens SUFFIX .js RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/web/public/wasm")
  target_link_options(tracelens_wasm PRIVATE --bind -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=web,worker,node -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=268435456 -sFILESYSTEM=0)
else()
  add_executable(tracelens core/cli.cpp)
  target_link_libraries(tracelens PRIVATE tracelens_core)
  enable_testing()
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE tracelens_core)
  target_compile_definitions(engine_tests PRIVATE SAMPLE_PATH="${CMAKE_SOURCE_DIR}/samples/checkout-outage.tsv")
  add_test(NAME engine_contract COMMAND engine_tests)
endif()
